\begin{frame}{The principle}
\Large Every result carries what is needed to judge it.
\vspace{1em}
\normalsize
\begin{itemize}
\item a kernel: evidence that it computes the oracle's bits, on every substrate
\item an optimal integer program: the branch-and-bound tree, checked by separate code
\item a refuted conservation law: the polynomial remainder and the point where it is not zero
\item a statistical claim: a control that a broken implementation would have passed
\end{itemize}
\vspace{1em}
Three properties: \textbf{determinism}, \textbf{certification}, \textbf{containment}.
\end{frame}

\begin{frame}{Programs are terms; meaning is the oracle's}
\begin{itemize}
\item a symbolic free algebra: element-wise ops, \texttt{sel}, contractions (\texttt{linear}, grouped, masked, 4-bit, int8), reductions, \texttt{gather\_row}, \texttt{rope}, \texttt{kv\_write}, \texttt{attention}, \texttt{sample}
\item semi-dynamic dimensions: bounds at the maximum, extended by monotonicity lemmas (Lean)
\item recurrent programs: the loop runs in the worker; the state never leaves it
\item \texttt{Vapor.F32}: exact binary32 on the BEAM (fma in dyadic rationals, one rounding)
\end{itemize}
\vspace{0.6em}
\begin{tabularx}{\linewidth}{@{}lX@{}}
\toprule
\texttt{:canonical} & separate rounding; one 16-lane reduction tree; $\div$ correctly rounded $\Rightarrow$ \textbf{identical bits} on every substrate \\
\texttt{:fast} & fused FMA $\Rightarrow$ every output inside a \textbf{rigorous envelope} (exact dyadic bounds) \\
\bottomrule
\end{tabularx}

\vspace{0.6em}
Parallelism from \textbf{independent outputs}, never from reassociating a sum.
\end{frame}

  \foreach \a/\b in {t/r,r/l,l/k,k/sel,sel/ra,ra/chk,chk/enc} \draw[edge] (\a)--(\b);
\end{tikzpicture}

\vspace{0.8em}
\begin{itemize}
\item rewriting admits only identities valid over all of IEEE-754 ($x+(+0) \not\to x$)
\item one kernel IR over typed virtual registers; $g$ generalises LMUL to AVX2, AVX-512, NEON, RVV
\item out of registers? smaller $g$, fewer rows, or cut the fused region: \textbf{never spill}
\item every allocation revalidated by \texttt{check\_alloc}, extracted from Lean with \texttt{checkAlloc\_sound}
\item encoders write bytes (REX/VEX/EVEX, A64, RV64GCV, SPIR-V); binutils and \texttt{spirv-val} only check
\end{itemize}
\end{frame}

\begin{frame}{Substrates and isolation}
\begin{columns}[T]
\column{0.5\linewidth}
\textbf{vapor-worker} (static Zig, no libc, 350\,KB)
\begin{itemize}\small
\item seccomp allow-list, \texttt{NO\_NEW\_PRIVS}, W\^{}X pages, watchdog
\item thread pool and counters set up before the filter
\item native code, or the RVV~1.0 interpreter (VLEN 128--512, poison mode)
\item resident sessions: weights mapped once, state kept
\end{itemize}
\textbf{vapor-fabric}: Vulkan compute, its own process
\column{0.5\linewidth}
\textbf{One protocol}: \texttt{\{packet,4\}} frames on stdio
\begin{itemize}\small
\item \textsc{hello} $\cdot$ \textsc{run} $\cdot$ \textsc{emit} $\cdot$ \textsc{done} $\cdot$ \textsc{err} $\cdot$ \textsc{open}/\textsc{step}/\textsc{close}
\item a crash kills only the process; the next unit runs
\end{itemize}
\textbf{Memory}: weights in \texttt{/dev/shm/vapor-<sha256>}, mapped copy-on-write; imported without copy by Vulkan
\textbf{Failover}: fabric $\to$ AVX-512 $\to$ base $\to$ RVV interpreter $\to$ oracle: same bits at every link
\end{columns}
\end{frame}

\begin{frame}{The verification ladder}
\small
\begin{tabularx}{\linewidth}{@{}clX@{}}
\toprule
1 & sorts, shapes & \texttt{Program.check/1} \\
2 & admission & checker accepts the allocation on \emph{every} ISA; no int8 wrap at the max extents \\
3 & adjoint & $\langle Wx,v\rangle = \langle x,W^{\mathsf T}v\rangle$ catches layout errors \\
4 & oracle & each CPU substrate = exact oracle, bit for bit \\
5 & parity, envelope & equal digests across substrates; outputs inside the envelope \\
6 & certificate & Ed25519; host-independent payload $\Rightarrow$ byte-identical across nodes $\Rightarrow$ co-signable, quorum \\
\bottomrule
\end{tabularx}

\vspace{0.6em}
\textbf{Lean~4} (core only, no \texttt{sorry}/\texttt{axiom}): integer parity of any reduction tree, bank injectivity, the affine monoid, allocation soundness, envelope, Higham's bound $\to$ extracted at build time, 480+ conformance vectors.
\end{frame}

\begin{frame}{A frontier topology: the delta-rule hybrid}
\small
\[
S_t = (I - \beta_t k_t k_t^{\mathsf T})\,\mathrm{Diag}(\alpha_t)\,S_{t-1} + \beta_t k_t v_t^{\mathsf T},
\qquad \alpha_t = e^{g_t},\ g_t = g_{\min}\sigma(e^{A}z_t)
\]
\vspace{-0.4em}
\begin{itemize}
\item chunked prefill via the UT transform $T = (I + \mathrm{diag}(\beta)L_s)^{-1}$: equal to the recurrence to $3\cdot10^{-16}$
\item latent attention: only the latent cached; key map absorbed into the query
\item latent mixture of experts; soft-capped gated activation; \textbf{MXFP4 decodes exactly} to binary32
\item vs an independent float64 reference: rel.\ error $4.4\cdot10^{-6}$; greedy identical; native = oracle; controls fail
\item balanced routing is a totally unimodular $b$-matching: \textbf{certified} by the exact simplex
\end{itemize}
\end{frame}

\begin{frame}{Containment of open input}
\begin{columns}[T]
\column{0.5\linewidth}
\textbf{Alembic}: problems as text
\begin{itemize}\small
\item fuel on every call and iteration; bounded depth
\item integers $\le$ 65{,}536 bits; lists $\le$ 2 million
\item no atoms created; no I/O, clock or randomness
\item a literal reader for data
\end{itemize}
\textbf{Athanor}: portfolio search with a random control; journal under a Merkle root; the \textbf{Touchstone} re-checks every certificate
\column{0.5\linewidth}
\textbf{The hermetic seal}
\begin{itemize}\small
\item one primitive for every job on untrusted input
\item heap cap that \emph{counts off-heap binaries}
\item deadline; the caller survives a memory bomb
\end{itemize}
\vspace{0.5em}
\textbf{Agents}: effect classes (pure, observe, act), grants in the digest, write-ahead journal, replay is verification
\end{columns}
\end{frame}

\begin{frame}{The evidence discipline}
\small
Every check = \textbf{value} + \textbf{control} + \textbf{threshold}: \emph{would this answer come out with nothing to find?}

\begin{tabularx}{\linewidth}{@{}lXX@{}}
\toprule

\midrule

ODE & Dormand--Prince $10^{-8}$ & RK4 $h{=}0.1$: $>10^{-5}$ \\

\begin{frame}{Limits}
\small
\begin{itemize}
\item RVV and NEON validated under QEMU, Vulkan under lavapipe; no physical RVV, discrete GPU or Apple Silicon run
\item transcendental functions identical everywhere but within 1--2 ulps, not correctly rounded
\item 4-bit GEMV bound by instruction issue; the Vulkan path is correct, not tuned
\item frontier topologies checked on small instances; released spellings read through header-only admission
\item hosted models are observations, not functions
\item a certificate says what was computed, not whether a model is good; a causal verdict needs its diagram
\item the seal bounds memory and time, not code with access to the BEAM
\end{itemize}
\end{frame}
